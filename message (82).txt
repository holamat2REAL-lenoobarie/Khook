-- ─── Kick ─────────────────────────────────────────────────────────────
local function psCode(link)
    link = tostring(link or ""):match("^%s*(.-)%s*$")
    if link == "" then return nil end
    if not link:find("://", 1, true) then return link end
    return link:match("[?&]privateServerLinkCode=([^&]+)")
        or link:match("[?&]linkCode=([^&]+)")
        or link:match("[?&]code=([^&]+)")
end

-- toPS: only the auto kick on steal sends you to the private server
local function doKick(toPS)
    if toPS == true and _G.MeerkoKickToPS == true then
        local code = psCode(_G.MeerkoPrivateServerLink)
        if code and code ~= "" then
            local ok = pcall(function()
                game:GetService("ExperienceService"):LaunchExperience({
                    placeId  = tonumber(_G.MeerkoPrivateServerPlaceId) or game.PlaceId,
                    linkCode = code,
                })
            end)
            if ok then return end
        end
    end
    -- SetErrorMessage affiche l'overlay Roblox natif avec le message.
    -- LP:Kick ferme la connexion — le jeu reste visible derrière l'overlay,
    -- pas de retour au menu.
    pcall(function()
        game:GetService("GuiService"):SetErrorMessage("I LOVE BR4DO")
    end)
    task.wait(0.05)
    pcall(function() LP:Kick("I LOVE BR4DO") end)
    -- auto-close the kick box (no Leave click); the prompt can show up late, so keep clearing for ~3s
    for _ = 1, 30 do
        task.wait(0.1)
        pcall(function() game:GetService("GuiService"):ClearError() end)
        pcall(function()
            local overlay = game:GetService("CoreGui").RobloxPromptGui.promptOverlay
            for _, c in ipairs(overlay:GetChildren()) do c:Destroy() end
        end)
    end
end
_G.MeerkoKickOut = doKick

-- ─── Auto Kick — watcher texte "you stole" (signal de dépôt en base) ─
-- Même logique qu'extras_2/Mynxx : on écoute le texte GUI que le serveur
-- affiche APRÈS que le brainrot est déposé en base. C'est le seul signal
-- côté client qui confirme un steal COMPLET. L'attribut Stealing repasse
-- à false aussi quand on lâche le brainrot des mains — on l'ignore.
-- Flag "fired" : un kick par steal, réarmé après respawn.

if _G.MeerkoAutoKickOnSteal == nil then _G.MeerkoAutoKickOnSteal = false end
local KEYWORD = "you stole"
local fired   = false

local function checkText(txt)
    if not _G.MeerkoAutoKickOnSteal then return end
    if type(txt) ~= "string" or txt == "" then return end
    if fired then return end
    if string.find(string.lower(txt), KEYWORD, 1, true) then
        fired = true
        task.spawn(doKick, true)
    end
end

local hooked = setmetatable({}, { __mode = "k" })
local function hookObj(obj)
    if hooked[obj] then return end
    hooked[obj] = true
    checkText(obj.Text)
    obj:GetPropertyChangedSignal("Text"):Connect(function()
        checkText(obj.Text)
    end)
end

local function isText(o)
    return o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox")
end

local function watchRoot(root)
    for _, obj in ipairs(root:GetDescendants()) do
        if isText(obj) then hookObj(obj) end
    end
    root.DescendantAdded:Connect(function(desc)
        if isText(desc) then hookObj(desc) end
    end)
end

task.spawn(function()
    local PG = LP:FindFirstChildOfClass("PlayerGui") or LP:WaitForChild("PlayerGui", 10)
    if not PG then return end
    for _, g in ipairs(PG:GetChildren()) do pcall(watchRoot, g) end
    PG.ChildAdded:Connect(function(g) pcall(watchRoot, g) end)
end)

-- Réarmer fired après respawn
LP.CharacterAdded:Connect(function()
    task.delay(1, function() fired = false end)
end)

-- ─── Script Exec ──────────────────────────────────────────────────────
local function loadSavedScript()
    if readfile then
        local ok, src = pcall(readfile, SCRIPT_FILE)
        if ok and type(src) == "string" and src:match("%S") then return src end
    end
    return ""
end

local function saveScript(src)
    if not writefile then return false end
    return pcall(writefile, SCRIPT_FILE, src)
end

local currentScript = loadSavedScript()

local function execScript()
    if not currentScript:match("%S") then return false, "empty" end
    local chunk, err = loadstring(currentScript)
    if not chunk then return false, err end
    local ok, err2 = pcall(chunk)
    return ok, err2
end

-- ── Keybind picker générique ───────────────────────────────────────────
-- Crée un panneau de changement de touche rattaché à un bouton.
-- Paramètres :
--   btn        : TextButton cible (clic droit pour ouvrir)
--   getKey()   : retourne la keybind actuelle (string)
--   setKey(k)  : enregistre la nouvelle keybind
--   getLabel() : retourne le texte à afficher sur le bouton
--   updateLabel() : met à jour le texte du bouton
local function makeKeybindPicker(btn, getKey, setKey, updateLabel)
    local KP_W, KP_H = 180, 68
    local kp = Instance.new("Frame", sg)
    kp.Size             = UDim2.fromOffset(KP_W, KP_H)
    kp.BackgroundColor3 = Color3.fromRGB(8, 8, 8)
    kp.BorderSizePixel  = 0
    kp.Visible          = false
    kp.ZIndex           = 30
    Instance.new("UICorner", kp).CornerRadius = UDim.new(0, 8)
    local kpS = Instance.new("UIStroke", kp)
    kpS.Color = ICE_BLUE; kpS.Thickness = 1

    local kpTitle = Instance.new("TextLabel", kp)
    kpTitle.Size                   = UDim2.new(1, -12, 0, 20)
    kpTitle.Position               = UDim2.fromOffset(6, 6)
    kpTitle.BackgroundTransparency = 1
    kpTitle.Text                   = "APPUIE UNE TOUCHE…"
    kpTitle.TextColor3             = ICE_BLUE
    kpTitle.Font                   = Enum.Font.GothamBold
    kpTitle.TextSize               = 10
    kpTitle.TextXAlignment         = Enum.TextXAlignment.Center
    kpTitle.ZIndex                 = 31

    local kpCurrent = Instance.new("TextLabel", kp)
    kpCurrent.Size                   = UDim2.new(1, -12, 0, 22)
    kpCurrent.Position               = UDim2.fromOffset(6, 26)
    kpCurrent.BackgroundTransparency = 1
    kpCurrent.TextColor3             = Color3.fromRGB(180, 180, 180)
    kpCurrent.Font                   = Enum.Font.GothamBold
    kpCurrent.TextSize               = 10
    kpCurrent.TextXAlignment         = Enum.TextXAlignment.Center
    kpCurrent.ZIndex                 = 31

    local kpCancel = Instance.new("TextButton", kp)
    kpCancel.Size             = UDim2.new(1, -12, 0, 16)
    kpCancel.Position         = UDim2.fromOffset(6, 48)
    kpCancel.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    kpCancel.BorderSizePixel  = 0
    kpCancel.AutoButtonColor  = false
    kpCancel.Text             = "ANNULER"
    kpCancel.TextColor3       = Color3.fromRGB(150, 150, 150)
    kpCancel.Font             = Enum.Font.GothamBold
    kpCancel.TextSize         = 9
    kpCancel.ZIndex           = 31
    Instance.new("UICorner", kpCancel).CornerRadius = UDim.new(0, 4)

    local listening = false
    local conn

    local function close()
        listening = false
        kp.Visible = false
        if conn then conn:Disconnect(); conn = nil end
        kpTitle.Text      = "APPUIE UNE TOUCHE…"
        kpTitle.TextColor3 = ICE_BLUE
    end

    local function open()
        local bpos = btn.AbsolutePosition
        local fp   = frame.AbsolutePosition
        local fs   = frame.AbsoluteSize
        local cam  = workspace.CurrentCamera
        local view = cam and cam.ViewportSize or Vector2.new(1280, 720)
        local x = math.clamp(math.floor(bpos.X), 6, math.max(6, view.X - KP_W - 6))
        local y = math.clamp(math.floor(fp.Y + fs.Y + 6), 6, math.max(6, view.Y - KP_H - 6))
        kp.Position = UDim2.fromOffset(x, y)
        kpCurrent.Text = "Actuel : " .. getKey()
        kp.Visible = true
        listening  = true
        conn = UIS.InputBegan:Connect(function(input, _gpe)
            if not listening then return end
            if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
            local k = tostring(input.KeyCode.Name)
            if k == "Escape" then close() return end
            if k == "Unknown" then return end
            setKey(k)
            updateLabel()
            kpCurrent.Text     = "Actuel : " .. k
            kpTitle.Text       = "✓ " .. k .. " enregistré!"
            kpTitle.TextColor3 = Color3.fromRGB(60, 200, 100)
            task.delay(1, close)
        end)
    end

    kpCancel.MouseButton1Click:Connect(close)
    btn.MouseButton2Click:Connect(function()
        if kp.Visible then close() else open() end
    end)
end

-- Logique extraite exactement du script sabcom_flasher.
local FLASH_ANIM_ID = "70883871260184"
local antiCfg       = { anti = false }
local antiConns     = {}
local antiToken     = 0
local origMesh      = setmetatable({}, { __mode = "k" })
local donorMap

local function getAntiSave()
    if not readfile then return end
    local ok, d = pcall(function()
        return game:GetService("HttpService"):JSONDecode(readfile("medzna_antiflash.json"))
    end)
    if ok and type(d) == "table" and type(d.anti) == "boolean" then
        antiCfg.anti = d.anti
    end
end
local function saveAntiCfg()
    pcall(function()
        if writefile then
            writefile("medzna_antiflash.json",
                game:GetService("HttpService"):JSONEncode(antiCfg))
        end
    end)
end
getAntiSave()

local function getDonor()
    if donorMap then return donorMap end
    donorMap = {}
    pcall(function()
        local r = Players:CreateHumanoidModelFromDescription(
            Instance.new("HumanoidDescription"), Enum.HumanoidRigType.R15)
        for _, d in ipairs(r:GetChildren()) do
            if d:IsA("MeshPart") then donorMap[d.Name] = d.MeshId end
        end
        r:Destroy()
    end)
    return donorMap
end

local function isFlashTrack(track)
    local id = ""
    pcall(function()
        local anim = track.Animation
        if anim then id = tostring(anim.AnimationId) end
    end)
    return id ~= "" and string.find(id, FLASH_ANIM_ID, 1, true) ~= nil
end

local function unskin(c)
    if not c then return end
    local mp = getDonor()
    for _, d in ipairs(c:GetDescendants()) do
        if d:IsA("MeshPart") then
            pcall(function()
                local skinned = d.HasSkinnedMesh
                if skinned then
                    if origMesh[d] == nil then origMesh[d] = d.MeshId end
                    if mp[d.Name] then d.MeshId = mp[d.Name] end
                    d.HasSkinnedMesh = false
                end
            end)
        elseif d:IsA("WrapLayer") or d:IsA("WrapTarget") then
            pcall(function() d:Destroy() end)
        end
    end
end

local function stopOtherFlash(c, owner)
    if owner == LP then return end
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    local anm = hum and hum:FindFirstChildOfClass("Animator")
    if not anm then return end
    for _, track in ipairs(anm:GetPlayingAnimationTracks()) do
        if isFlashTrack(track) then pcall(function() track:Stop(0) end) end
    end
end

local function harden(player)
    local c = player and player.Character
    if not c then return end
    unskin(c)
    stopOtherFlash(c, player)
end

local function reskin()
    for d, id in pairs(origMesh) do
        pcall(function() d.MeshId = id; d.HasSkinnedMesh = true end)
    end
    table.clear(origMesh)
end

local function stopAnti()
    antiToken += 1
    for i = 1, #antiConns do
        pcall(function() antiConns[i]:Disconnect() end)
    end
    table.clear(antiConns)
    reskin()
end

local function startAnti()
    stopAnti()
    antiCfg.anti = true
    local token = antiToken
    local function bind(player)
        if not player then return end
        antiConns[#antiConns + 1] = player.CharacterAdded:Connect(function(c)
            if not antiCfg.anti or token ~= antiToken then return end
            task.defer(harden, player)
            antiConns[#antiConns + 1] = c.DescendantAdded:Connect(function(d)
                if not antiCfg.anti or token ~= antiToken then return end
                if d:IsA("MeshPart") or d:IsA("WrapLayer") or d:IsA("WrapTarget") then
                    task.defer(unskin, c)
                elseif d:IsA("Animator") and player ~= LP then
                    antiConns[#antiConns + 1] = d.AnimationPlayed:Connect(function(track)
                        if antiCfg.anti and isFlashTrack(track) then
                            pcall(function() track:Stop(0) end)
                        end
                    end)
                end
            end)
        end)
        if player.Character then
            harden(player)
            local c = player.Character
            antiConns[#antiConns + 1] = c.DescendantAdded:Connect(function(d)
                if not antiCfg.anti or token ~= antiToken then return end
                if d:IsA("MeshPart") or d:IsA("WrapLayer") or d:IsA("WrapTarget") then
                    task.defer(unskin, c)
                end
            end)
            local hum = c:FindFirstChildOfClass("Humanoid")
            local anm = hum and hum:FindFirstChildOfClass("Animator")
            if anm and player ~= LP then
                antiConns[#antiConns + 1] = anm.AnimationPlayed:Connect(function(track)
                    if antiCfg.anti and isFlashTrack(track) then
                        pcall(function() track:Stop(0) end)
                    end
                end)
            end
        end
    end
    for _, player in ipairs(Players:GetPlayers()) do bind(player) end
    antiConns[#antiConns + 1] = Players.PlayerAdded:Connect(bind)
    task.spawn(function()
        while antiCfg.anti and token == antiToken do
            for _, player in ipairs(Players:GetPlayers()) do harden(player) end
            task.wait(0.05)
        end
    end)
end
